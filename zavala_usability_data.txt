-- Zavala County Website Redesign: Usability Test Data
-- PostgreSQL schema and data load
-- Source: moderated task-based usability test, 5 participants,
-- original site (co.zavala.tx.us) vs. redesigned site

-- Schema

CREATE SCHEMA IF NOT EXISTS Zavala;

SET search_path TO Zavala;

CREATE TABLE participants (
    participant_id      SMALLINT PRIMARY KEY,
    prior_experience     TEXT,          -- pre-task Q1: prior use of a county/city government site
    languages_comfortable TEXT,         -- pre-task Q2: languages read comfortably
    other_languages      TEXT,          -- pre-task Q2: other languages proficient in
    connection_speed     TEXT           -- pre-task Q3: typical internet connection
);

CREATE TABLE tasks (
    task_id                 SMALLINT PRIMARY KEY,
    task_number              SMALLINT NOT NULL,
    task_name                 TEXT NOT NULL,
    task_prompt                TEXT NOT NULL,
    stress_point                TEXT,
    design_decision_tested       TEXT,
    administered                 BOOLEAN NOT NULL DEFAULT TRUE  -- FALSE for Task 5, struck through in protocol, never run
);

CREATE TABLE task_results (
    result_id            SERIAL PRIMARY KEY,
    participant_id         SMALLINT NOT NULL REFERENCES participants(participant_id),
    task_id                 SMALLINT NOT NULL REFERENCES tasks(task_id),
    website_version           TEXT NOT NULL CHECK (website_version IN ('original', 'redesigned')),
    raw_time_text               TEXT,          -- time exactly as recorded on the stopwatch (mm:ss:hundredths)
    time_seconds                 NUMERIC(8,2), -- parsed time in seconds; NULL where the source value could not be parsed
    found_target                  TEXT CHECK (found_target IN ('yes', 'no', 'not_sure')),
    knew_next_step                  BOOLEAN,     -- TRUE = no hesitation reported; FALSE = participant paused/backtracked
    next_step_detail                  TEXT,      -- moderator note on where the participant paused or backtracked
    readability                        TEXT CHECK (readability IN ('easy', 'okay', 'hard')),
    would_give_up                       BOOLEAN,
    give_up_detail                        TEXT,
    notes                                   TEXT,  -- readability/UX comments not captured elsewhere
    data_quality_flag                        TEXT, -- ambiguity or transcription issue in the source document
    UNIQUE (participant_id, task_id, website_version)
);

CREATE TABLE cold_start_impressions (
    impression_id       SERIAL PRIMARY KEY,
    participant_id         SMALLINT NOT NULL REFERENCES participants(participant_id),
    website_version           TEXT NOT NULL CHECK (website_version IN ('original', 'redesigned')),
    impression                 TEXT NOT NULL,
    UNIQUE (participant_id, website_version)
);

CREATE TABLE closing_debriefs (
    participant_id            SMALLINT PRIMARY KEY REFERENCES participants(participant_id),
    most_tedious_task            TEXT,
    confusing_elements             TEXT,
    comparison_to_prior_sites        TEXT,
    one_fix                            TEXT
);

CREATE INDEX idx_task_results_participant ON task_results(participant_id);
CREATE INDEX idx_task_results_task ON task_results(task_id);
CREATE INDEX idx_task_results_version ON task_results(website_version);
