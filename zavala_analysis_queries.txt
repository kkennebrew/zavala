-- Queries for the evaluation section
-- Run after loading zavala_usability_data.txt

-- 1. Mean completion time, original vs. redesigned, per task
--    (excludes rows with NULL time_seconds, i.e. unparseable source times)
SELECT
    t.task_number,
    t.task_name,
    tr.website_version,
    COUNT(*)                          AS n_timed,
    ROUND(AVG(tr.time_seconds), 2)    AS mean_seconds,
    ROUND(MIN(tr.time_seconds), 2)    AS min_seconds,
    ROUND(MAX(tr.time_seconds), 2)    AS max_seconds
FROM task_results tr
JOIN tasks t ON t.task_id = tr.task_id
WHERE tr.time_seconds IS NOT NULL
GROUP BY t.task_number, t.task_name, tr.website_version
ORDER BY t.task_number, tr.website_version;

-- 2. Overall mean time, original vs. redesigned, across all tasks
SELECT
    website_version,
    COUNT(*)                       AS n_timed,
    ROUND(AVG(time_seconds), 2)    AS mean_seconds
FROM task_results
WHERE time_seconds IS NOT NULL
GROUP BY website_version;

-- 3. Task success rate ("found_target = yes"), original vs. redesigned
SELECT
    t.task_number,
    t.task_name,
    tr.website_version,
    COUNT(*) FILTER (WHERE tr.found_target = 'yes')       AS succeeded,
    COUNT(*)                                                 AS attempted,
    ROUND(100.0 * COUNT(*) FILTER (WHERE tr.found_target = 'yes') / COUNT(*), 1) AS success_pct
FROM task_results tr
JOIN tasks t ON t.task_id = tr.task_id
GROUP BY t.task_number, t.task_name, tr.website_version
ORDER BY t.task_number, tr.website_version;

-- 4. How often participants hesitated or backtracked (knew_next_step = FALSE)
SELECT
    website_version,
    COUNT(*) FILTER (WHERE knew_next_step = FALSE) AS hesitated,
    COUNT(*)                                          AS total,
    ROUND(100.0 * COUNT(*) FILTER (WHERE knew_next_step = FALSE) / COUNT(*), 1) AS hesitated_pct
FROM task_results
GROUP BY website_version;

-- 5. Readability ratings distribution, original vs. redesigned
SELECT
    website_version,
    readability,
    COUNT(*) AS n
FROM task_results
WHERE readability IS NOT NULL
GROUP BY website_version, readability
ORDER BY website_version, readability;

-- 6. Per-participant time saved (redesigned minus original) for tasks tested on both
SELECT
    o.participant_id,
    t.task_number,
    o.time_seconds AS original_seconds,
    r.time_seconds AS redesigned_seconds,
    ROUND(o.time_seconds - r.time_seconds, 2) AS seconds_saved
FROM task_results o
JOIN task_results r
    ON o.participant_id = r.participant_id
   AND o.task_id = r.task_id
   AND o.website_version = 'original'
   AND r.website_version = 'redesigned'
JOIN tasks t ON t.task_id = o.task_id
WHERE o.time_seconds IS NOT NULL AND r.time_seconds IS NOT NULL
ORDER BY o.participant_id, t.task_number;

-- 7. Rows flagged for data-quality review (ambiguous source times / labeling)
SELECT participant_id, task_id, website_version, raw_time_text, data_quality_flag
FROM task_results
WHERE data_quality_flag IS NOT NULL;

-- 8. "Would give up" rate, original vs. redesigned
SELECT
    website_version,
    COUNT(*) FILTER (WHERE would_give_up = TRUE) AS would_give_up,
    COUNT(*)                                        AS total,
    ROUND(100.0 * COUNT(*) FILTER (WHERE would_give_up = TRUE) / COUNT(*), 1) AS give_up_pct
FROM task_results
GROUP BY website_version;
